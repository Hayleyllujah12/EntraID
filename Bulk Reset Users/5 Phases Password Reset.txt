# Strip invisible Unicode formatting chars from pasted strings
function Remove-InvisibleChars {
    param([string]$s)
    if ([string]::IsNullOrEmpty($s)) { return $s }
    return ($s -replace '[\u200B-\u200F\u202A-\u202E\u2066-\u2069\uFEFF]', '')
}

# Sanitize a UPN: strip invisible/whitespace chars, NFD-normalize accents, lowercase, validate
function Sanitize-Upn {
    param([string]$Raw)

    if ([string]::IsNullOrEmpty($Raw)) {
        return [PSCustomObject]@{ Clean = ''; Changed = $false; Valid = $false; Reason = 'Empty' }
    }

    # 1. Strip invisible formatting + control chars
    $s = $Raw -replace '[\u200B-\u200F\u202A-\u202E\u2066-\u2069\uFEFF\u0000-\u001F\u007F]', ''

    # 2. Strip Unicode space variants (NBSP, narrow NBSP, hair spaces, etc.)
    $s = $s -replace '[\u00A0\u1680\u2000-\u200A\u202F\u205F\u3000]', ''

    # 3. Trim regular whitespace + quotes
    $s = $s.Trim().Trim('"').Trim("'").Trim()

    # 4. NFD normalize (é -> e + combining acute) then drop combining marks -> ASCII fold
    $normalized = $s.Normalize([Text.NormalizationForm]::FormD)
    $sb = [Text.StringBuilder]::new()
    foreach ($ch in $normalized.ToCharArray()) {
        if ([Globalization.CharUnicodeInfo]::GetUnicodeCategory($ch) -ne [Globalization.UnicodeCategory]::NonSpacingMark) {
            [void]$sb.Append($ch)
        }
    }
    $s = $sb.ToString().ToLowerInvariant()

    # 5. Validate
    $changed = ($s -ne $Raw)
    $valid   = $true
    $reason  = 'OK'
    if ($s -notmatch '^[^@\s]+@[^@\s]+\.[^@\s]+$') { $valid = $false; $reason = 'Invalid UPN format' }
    elseif ($s.Length -gt 113)                     { $valid = $false; $reason = 'Exceeds 113-char Entra limit' }

    return [PSCustomObject]@{
        Clean   = $s
        Changed = $changed
        Valid   = $valid
        Reason  = $reason
    }
}

# Graph call with exponential backoff + Retry-After header support
function Invoke-GraphWithRetry {
    param(
        [scriptblock]$ScriptBlock,
        [int]$MaxAttempts = 5,
        [int]$InitialDelaySeconds = 2
    )
    $attempt = 0
    $delay   = $InitialDelaySeconds
    while ($true) {
        $attempt++
        try {
            return & $ScriptBlock
        }
        catch {
            $ex     = $_.Exception
            $status = $null
            $retryAfter = $null

            # Extract HTTP status + Retry-After header if available
            if ($ex.PSObject.Properties.Name -contains 'Response' -and $ex.Response) {
                try { $status = [int]$ex.Response.StatusCode } catch {}
                try {
                    $ra = $ex.Response.Headers['Retry-After']
                    if ($ra) { $retryAfter = [int]$ra }
                } catch {}
            }

            $isRetryable = ($status -in 429, 500, 502, 503, 504) -or
                           ($ex.Message -match 'throttl|timeout|temporarily|service unavailable')

            if (-not $isRetryable -or $attempt -ge $MaxAttempts) {
                throw
            }

            $wait = if ($retryAfter) { $retryAfter } else { $delay }
            Write-Host "    Retry $attempt/$MaxAttempts after $wait s (status=$status)" -ForegroundColor DarkYellow
            Start-Sleep -Seconds $wait
            $delay = [Math]::Min($delay * 2, 60)
        }
    }
}

# SHA256 hash of a file (audit fingerprint of the input CSV)
function Get-FileHashSHA256 {
    param([string]$Path)
    try { return (Get-FileHash -Path $Path -Algorithm SHA256).Hash } catch { return 'HASH_ERROR' }
}

# Ensure a module is present + imported. Prompts (Y/N) before installing.
# Returns $true on success, $false on any failure so caller can `return`.
function Ensure-Module {
    param(
        [Parameter(Mandatory)][string]$Name,
        [switch]$AutoInstall  # bypass prompt if $true
    )

    # Already installed? Just import and continue.
    if (Get-Module -ListAvailable -Name $Name) {
        try {
            Import-Module $Name -ErrorAction Stop
            return $true
        }
        catch {
            Write-Host "ERROR: '$Name' is installed but failed to import: $($_.Exception.Message)" -ForegroundColor Red
            Write-Host "This often indicates Graph SDK version drift. Recommended clean reinstall:" -ForegroundColor Yellow
            Write-Host "  Uninstall-Module Microsoft.Graph -AllVersions -Force" -ForegroundColor Yellow
            Write-Host "  Get-InstalledModule Microsoft.Graph* -AllVersions | Uninstall-Module -Force" -ForegroundColor Yellow
            Write-Host "  Install-Module Microsoft.Graph -Scope CurrentUser -Force" -ForegroundColor Yellow
            return $false
        }
    }

    # Missing - prompt (never silent install)
    Write-Host ""
    Write-Host "Module '$Name' is not installed." -ForegroundColor Yellow
    $doInstall = $AutoInstall.IsPresent
    if (-not $doInstall) {
        $ans = Read-Host "Install '$Name' now for CurrentUser? (Y/N)"
        $doInstall = ($ans -match '^(y|yes)$')
    }
    if (-not $doInstall) {
        Write-Host "Install declined. Cannot continue without '$Name'." -ForegroundColor Red
        return $false
    }

    try {
        Write-Host "Installing '$Name' (Scope: CurrentUser)..." -ForegroundColor Cyan
        Install-Module -Name $Name -Scope CurrentUser -Force -AllowClobber -ErrorAction Stop
        Import-Module $Name -ErrorAction Stop
        $installed = Get-Module $Name | Select-Object -First 1
        Write-Host "  Installed and imported: $Name v$($installed.Version)" -ForegroundColor Green
        return $true
    }
    catch {
        Write-Host "ERROR: Install of '$Name' failed: $($_.Exception.Message)" -ForegroundColor Red
        Write-Host "Verify internet access, PSGallery trust, and permissions." -ForegroundColor Yellow
        return $false
    }
}

if ($ReadyCount -gt 500) {
    Write-Host ""
    Write-Host "Large batch detected ($ReadyCount rows)." -ForegroundColor Yellow
    $sizeConfirm = Read-Host "Type the exact count '$ReadyCount' to proceed"
    if ($sizeConfirm -ne $ReadyCount.ToString()) {
        Write-Host "Count mismatch. Cancelling." -ForegroundColor Red
        Disconnect-MgGraph -ErrorAction SilentlyContinue | Out-Null
        return
    }
}

Write-Host ""
$Confirm = Read-Host "Final confirmation - proceed with $ReadyCount password reset(s)? (Y/N)"
if ($Confirm -notmatch '^(y|yes)$') {
    Write-Host "Cancelled. No changes were made." -ForegroundColor Red
    Disconnect-MgGraph -ErrorAction SilentlyContinue | Out-Null
    return
}
Write-Host "Confirmed. Proceeding..." -ForegroundColor Green


# ============================================================
# PHASE 5 - Execute + Flush + Export Audit
# ============================================================
Write-PhaseHeader "PHASE 5: Reset Passwords + Export Audit Log"

$RunStart = Get-Date
$Log      = [System.Collections.Generic.List[object]]::new()
$Counters = @{ Success = 0; Failed = 0; Skipped = 0 }
$RowIndex = 0
$Total    = $ProcessedRows.Count

# Periodic flush helper
$FlushEvery = 100
function Flush-Log {
    param([System.Collections.Generic.List[object]]$Log, [string]$Path)
    try { $Log | Export-Csv -Path $Path -NoTypeInformation -Encoding UTF8 -Force -ErrorAction Stop } catch { }
}

foreach ($row in $ProcessedRows) {
    $RowIndex++
    $rowStart = Get-Date
    $status   = 'Failed'
    $message  = ''
    $graphErr = ''
    $httpCode = ''
    $retries  = 0
    $userId   = ''

    # Skip anything not marked Ready by pre-flight
    if ($row.PreflightStatus -ne 'Ready') {
        $status  = 'Skipped'
        $message = $row.PreflightMessage
        Write-Host "[Row $RowIndex/$Total] SKIP  $($row.UpnClean) - $message" -ForegroundColor Yellow
        $Counters.Skipped++
    }
    else {
        try {
            Write-Host "[Row $RowIndex/$Total] Reset $($row.UpnClean)..." -ForegroundColor Cyan
            $PasswordProfile = @{
                Password                      = $row.Password
                ForceChangePasswordNextSignIn = $true
            }
            $capturedRetries = 0
            Invoke-GraphWithRetry -ScriptBlock {
                $script:capturedRetries++
                Update-MgUser -UserId $row.UpnClean -PasswordProfile $PasswordProfile -ErrorAction Stop
            } | Out-Null
            $retries = [Math]::Max(0, $capturedRetries - 1)
            $status  = 'Success'
            $message = 'Password reset successful'
            $Counters.Success++
            Write-Host "[Row $RowIndex/$Total] OK    $($row.UpnClean)" -ForegroundColor Green
        }
        catch {
            $status  = 'Failed'
            $message = $_.Exception.Message
            if ($_.ErrorDetails -and $_.ErrorDetails.Message) {
                $message = "$message | Details: $($_.ErrorDetails.Message)"
            }
            try { $graphErr = $_.Exception.Response.Headers['x-ms-ags-diagnostic'] } catch {}
            try { $httpCode = [int]$_.Exception.Response.StatusCode } catch {}
            $Counters.Failed++
            Write-Host "[Row $RowIndex/$Total] FAIL  $($row.UpnClean) - $message" -ForegroundColor Red
        }
    }

    $durationMs = [int]((Get-Date) - $rowStart).TotalMilliseconds

    $Log.Add([PSCustomObject]@{
        TimestampLocal      = (Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
        TimestampUtc        = (Get-Date).ToUniversalTime().ToString('o')
        Row                 = $row.Row
        UpnRaw              = $row.UpnRaw
        UpnCleaned          = $row.UpnClean
        SanitizationApplied = if ($row.UpnChanged) { 'Yes' } else { 'No' }
        UserObjectId        = $userId
        TemporaryPassword   = $row.Password
        Status              = $status
        Message             = $message
        HttpStatusCode      = $httpCode
        GraphErrorCode      = $graphErr
        RetryCount          = $retries
        DurationMs          = $durationMs
        OperatorAccount     = $OperatorAccount
        TenantId            = $TenantId
        ScriptVersion       = $ScriptVersion
    })

    # Heartbeat every 25 rows
    if (($RowIndex % 25) -eq 0) {
        Write-Host ("  -- Progress: {0}/{1} | Success: {2} | Failed: {3} | Skipped: {4}" -f `
            $RowIndex, $Total, $Counters.Success, $Counters.Failed, $Counters.Skipped) -ForegroundColor Magenta
    }

    # Periodic flush every 100 rows
    if (($RowIndex % $FlushEvery) -eq 0) {
        Flush-Log -Log $Log -Path $LogPath
        Write-Host ("  -- Log flushed to disk ({0} rows)" -f $Log.Count) -ForegroundColor DarkGray
    }
}

# Final flush of detail log
try {
    $Log | Export-Csv -Path $LogPath -NoTypeInformation -Encoding UTF8 -Force -ErrorAction Stop
    Write-Host ""
    Write-Host "Detail log saved: $LogPath" -ForegroundColor Green
}
catch {
    Write-Host "ERROR: Failed to export detail log: $($_.Exception.Message)" -ForegroundColor Red
}

# Failed-only CSV (schema matches input for easy re-run)
$FailedRows = $Log | Where-Object { $_.Status -eq 'Failed' }
if ($FailedRows.Count -gt 0) {
    try {
        $FailedRows | Select-Object @{n='upn';e={$_.UpnCleaned}}, @{n='password';e={$_.TemporaryPassword}} |
            Export-Csv -Path $FailedLogPath -NoTypeInformation -Encoding UTF8 -Force -ErrorAction Stop
        Write-Host "Failed-only CSV (re-run ready): $FailedLogPath" -ForegroundColor Green
    } catch {
        Write-Host "WARNING: Failed to export failed-only CSV: $($_.Exception.Message)" -ForegroundColor Yellow
    }
}

# Run manifest JSON
$RunEnd     = Get-Date
$DurationS  = [int]($RunEnd - $RunStart).TotalSeconds
$Manifest   = [ordered]@{
    ScriptVersion   = $ScriptVersion
    StartTimeLocal  = $RunStart.ToString('yyyy-MM-dd HH:mm:ss')
    EndTimeLocal    = $RunEnd.ToString('yyyy-MM-dd HH:mm:ss')
    StartTimeUtc    = $RunStart.ToUniversalTime().ToString('o')
    EndTimeUtc      = $RunEnd.ToUniversalTime().ToString('o')
    DurationSeconds = $DurationS
    Operator        = $OperatorAccount
    TenantId        = $TenantId
    Host            = $env:COMPUTERNAME
    PSVersion       = $PSVersionTable.PSVersion.ToString()
    GraphSdkVersion = (Get-Module Microsoft.Graph.Users | Select-Object -First 1 -ExpandProperty Version).ToString()
    Input           = [ordered]@{
        CsvPath   = $CsvPath
        CsvSha256 = $CsvHash
        RowCount  = $Total
    }
    Outputs         = [ordered]@{
        DetailLog  = $LogPath
        FailedLog  = if ($FailedRows.Count -gt 0) { $FailedLogPath } else { $null }
        Manifest   = $ManifestPath
    }
    Counts          = [ordered]@{
        Total            = $Total
        Success          = $Counters.Success
        Failed           = $Counters.Failed
        Skipped          = $Counters.Skipped
        PreflightHybrid  = $PreflightCounts.Hybrid
        PreflightNotFound= $PreflightCounts.NotFound
        PreflightInvalid = $PreflightCounts.InvalidUpn
        PreflightBadPwd  = $PreflightCounts.BadPassword
        PreflightPriv    = $PreflightCounts.Privileged
        Sanitized        = $SanitizedCount
    }
}
try {
    $Manifest | ConvertTo-Json -Depth 5 | Out-File -FilePath $ManifestPath -Encoding UTF8 -Force
    Write-Host "Run manifest saved : $ManifestPath" -ForegroundColor Green
} catch {
    Write-Host "WARNING: Failed to write manifest: $($_.Exception.Message)" -ForegroundColor Yellow
}

# Summary
Write-Host ""
Write-Host "============================================================" -ForegroundColor Cyan
Write-Host " Summary" -ForegroundColor Cyan
Write-Host "------------------------------------------------------------" -ForegroundColor Cyan
Write-Host (" Total processed : {0}" -f $Total)
Write-Host (" Success         : {0}" -f $Counters.Success) -ForegroundColor Green
Write-Host (" Failed          : {0}" -f $Counters.Failed)  -ForegroundColor Red
Write-Host (" Skipped         : {0}" -f $Counters.Skipped) -ForegroundColor Yellow
Write-Host (" Duration        : {0}s" -f $DurationS)
Write-Host (" End Time        : {0}" -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'))
Write-Host "============================================================" -ForegroundColor Cyan

# Disconnect
try {
    Disconnect-MgGraph -ErrorAction Stop | Out-Null
    Write-Host "Disconnected from Microsoft Graph." -ForegroundColor DarkGray
}
catch {
    Write-Host "Warning: Disconnect-MgGraph failed: $($_.Exception.Message)" -ForegroundColor Yellow
}
