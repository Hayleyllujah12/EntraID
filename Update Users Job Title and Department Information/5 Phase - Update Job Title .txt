    Bulk-update Entra ID user profile fields (First/Last/Display name, Job Title, Department)

    A whole row is skipped if nothing needs changing. Supports a DryRun (preview only) mode.

function Sanitize-Path {
    param([string]$Path)
    if ([string]::IsNullOrWhiteSpace($Path)) { return '' }
    $p = $Path.Trim().Trim('"').Trim("'")
    $invisible = @([char]0x00A0,[char]0x202A,[char]0x202B,[char]0x202C,[char]0x202D,
                   [char]0x202E,[char]0x200E,[char]0x200F,[char]0xFEFF,[char]0x200B)
    foreach ($c in $invisible) { $p = $p.Replace([string]$c,'') }
    return $p.Trim()
}

function Sanitize-Upn {
    param([Parameter(Mandatory)][AllowEmptyString()][string]$Value)

    if ([string]::IsNullOrWhiteSpace($Value)) {
        return @{ Clean=''; Changed=$true; Reason='Empty or whitespace-only' }
    }

    $raw = $Value
    # NFD normalise then strip combining marks (e-acute -> e, n-tilde -> n)
    $nfd = $raw.Normalize([Text.NormalizationForm]::FormD)
    $sb  = [Text.StringBuilder]::new()
    foreach ($ch in $nfd.ToCharArray()) {
        if ([Globalization.CharUnicodeInfo]::GetUnicodeCategory($ch) -ne [Globalization.UnicodeCategory]::NonSpacingMark) {
            [void]$sb.Append($ch)
        }
    }
    $s = $sb.ToString()

    $invisible = @(
        [char]0x00A0,[char]0x1680,[char]0x2000,[char]0x2001,[char]0x2002,[char]0x2003,
        [char]0x2004,[char]0x2005,[char]0x2006,[char]0x2007,[char]0x2008,[char]0x2009,
        [char]0x200A,[char]0x200B,[char]0x200C,[char]0x200D,[char]0x200E,[char]0x200F,
        [char]0x202A,[char]0x202B,[char]0x202C,[char]0x202D,[char]0x202E,
        [char]0x202F,[char]0x205F,[char]0x2060,[char]0x3000,[char]0xFEFF
    )
    foreach ($c in $invisible) { $s = $s.Replace([string]$c,'') }
    $s = $s -replace '\p{C}',''          # control chars
    $s = $s.Trim().ToLowerInvariant()

    $changed = ($s -ne $raw)
    $reason  = if ($changed) { 'Sanitized' } else { 'Unchanged' }

    if ($s -notmatch '^[^@\s]+@[^@\s]+\.[^@\s]+$') {
        return @{ Clean=$s; Changed=$changed; Reason='Invalid UPN format' }
    }
    if ($s.Length -gt 113) {
        return @{ Clean=$s; Changed=$changed; Reason='Exceeds 113-char Entra limit' }
    }
    return @{ Clean=$s; Changed=$changed; Reason=$reason }
}

function Ensure-Module {
    param([Parameter(Mandatory)][string]$Name, [string]$MinVersion)

    $existing = Get-Module -ListAvailable -Name $Name |
                Sort-Object Version -Descending | Select-Object -First 1

    if ($existing) {
        if ($MinVersion -and ($existing.Version -lt [version]$MinVersion)) {
            Write-Host "[WARN] $Name $($existing.Version) is older than required $MinVersion." -ForegroundColor Yellow
            $ans = Read-Host "Uninstall all versions and reinstall clean? (Y/N)"
            if ($ans -notmatch '^(y|yes)$') { return $false }
            try {
                Get-InstalledModule "$Name*" -AllVersions -ErrorAction SilentlyContinue |
                    Uninstall-Module -Force -ErrorAction SilentlyContinue
                Install-Module $Name -Scope CurrentUser -Force -AllowClobber -ErrorAction Stop
            } catch {
                Write-Host "[FAIL] Reinstall of $Name failed: $($_.Exception.Message)" -ForegroundColor Red
                return $false
            }
        }
        Import-Module $Name -ErrorAction Stop
        return $true
    }

    Write-Host "[MISSING] Module $Name is not installed." -ForegroundColor Yellow
    $ans = Read-Host "Install $Name from PSGallery now? (Y/N)"
    if ($ans -notmatch '^(y|yes)$') { return $false }
    try {
        Install-Module $Name -Scope CurrentUser -Force -AllowClobber -ErrorAction Stop
        Import-Module $Name -ErrorAction Stop
        return $true
    } catch {
        Write-Host "[FAIL] Install of $Name failed: $($_.Exception.Message)" -ForegroundColor Red
        return $false
    }
}

function Invoke-GraphWithRetry {
    param(
        [Parameter(Mandatory)][scriptblock]$Action,
        [int]$MaxAttempts = 5,
        [int]$BaseDelaySeconds = 2
    )
    for ($i = 1; $i -le $MaxAttempts; $i++) {
        try { return & $Action }
        catch {
            $status = $null; $retryAfter = $null
            $ex = $_.Exception
            # StrictMode-safe probe for an HTTP response object.
            $respProp = $ex.PSObject.Properties['Response']
            if ($respProp -and $respProp.Value) {
                $resp = $respProp.Value
                $scProp = $resp.PSObject.Properties['StatusCode']
                if ($scProp -and $null -ne $scProp.Value) { $status = [int]$scProp.Value }
                $hdrProp = $resp.PSObject.Properties['Headers']
                if ($hdrProp -and $hdrProp.Value) {
                    try {
                        $vals = $null
                        if ($hdrProp.Value.TryGetValues('Retry-After', [ref]$vals)) {
                            $retryAfter = [int]($vals | Select-Object -First 1)
                        }
                    } catch { }
                }
            }
            if ($null -eq $status) {
                if ($_.ErrorDetails.Message -match 'TooManyRequests' -or $ex.Message -match '\b429\b') { $status = 429 }
            }
            $isRetryable = $status -in 429,500,502,503,504
            if (-not $isRetryable -or $i -eq $MaxAttempts) { throw }
            $delay = if ($retryAfter) { $retryAfter } else { [int][Math]::Pow($BaseDelaySeconds, $i) }
            Write-Host "[RETRY $i/$MaxAttempts] HTTP $status - sleeping ${delay}s..." -ForegroundColor DarkYellow
            Start-Sleep -Seconds $delay
        }
    }
}

function Write-Log {
    param(
        [string]$Message,
        [ValidateSet('INFO','WARN','ERROR','SUCCESS','AUDIT','SECURITY')][string]$Level='INFO'
    )
    $ts   = Get-Date -Format 'yyyy-MM-dd HH:mm:ss.fff'
    $line = "$ts [$Level] [$Global:CorrelationId] $Message"
    $color = @{INFO='Gray';WARN='Yellow';ERROR='Red';SUCCESS='Green';AUDIT='Cyan';SECURITY='Magenta'}[$Level]
    Write-Host $line -ForegroundColor $color
    if ($Global:JsonLogFile) {
        $obj = [ordered]@{ ts=$ts; level=$Level; correlationId=$Global:CorrelationId; message=$Message }
        ($obj | ConvertTo-Json -Compress) | Add-Content -LiteralPath $Global:JsonLogFile -Encoding UTF8
    }
}

function Import-InputFile {
    param([Parameter(Mandatory)][string]$Path)
    $ext = [System.IO.Path]::GetExtension($Path).ToLowerInvariant()
    if ($ext -in '.xlsx','.xls','.xlsm') {
        if (-not (Ensure-Module -Name 'ImportExcel')) {
            throw "ImportExcel module is required to read '$ext' files but was not installed."
        }
        return @(Import-Excel -Path $Path)   # ImportExcel has no -LiteralPath; avoid bracketed paths for xlsx.
    }
    # Text file: sniff delimiter from the header line.
    $firstLine = Get-Content -LiteralPath $Path -TotalCount 1 -Encoding UTF8
    $delim = ','
    if ($firstLine -match "`t") { $delim = "`t" }
    elseif (($firstLine -split ';').Count -gt ($firstLine -split ',').Count) { $delim = ';' }
    return @(Import-Csv -LiteralPath $Path -Delimiter $delim -Encoding UTF8)
}

# Safe cell reader: returns $null if the header/column does not exist (avoids StrictMode throws).
function Get-Cell {
    param($Row, [string]$Column, [string[]]$Available)
    if ($Available -contains $Column) {
        $v = $Row.$Column
        if ($null -ne $v) { return [string]$v }
    }
    return $null
}

$plan      = [System.Collections.Generic.List[object]]::new()   # dry-run: previewed changes we can apply later

$dir = @{}
try {

foreach ($r in $rows) {

            Timestamp=$now; CorrelationId=$Global:CorrelationId; UserPrincipalName=$upn

    }

    # Build the update: only non-blank cells that differ from the current value (case-sensitive, trimmed).

    foreach ($f in $FieldMap) {

        $newRaw = Get-Cell -Row $r -Column $f.Csv -Available $headers

        if ($newVal -ceq $curVal) { continue }                            # already matches -> skip field
        $updateParams[$f.Graph] = $newVal
        $changeNotes.Add("$($f.Graph): '$curVal' -> '$newVal'")
    }

    if ($updateParams.Count -eq 0) {
        $skipC++
        $results.Add([pscustomobject]@{
            Timestamp=$now; CorrelationId=$Global:CorrelationId; UserPrincipalName=$upn
            Status='Skipped'; ChangedFields='None'; Changes=''; Message='No differing, non-blank fields'
        })
        if ($idx % $heartbeat -eq 0) { Write-Host "[HEARTBEAT] Row $idx/$total - OK $okC / FAIL $failC / SKIP $skipC / DRY $dryC" -ForegroundColor DarkGray }
        continue
    }

    $changedList = ($updateParams.Keys -join ', ')
    $changesStr  = ($changeNotes -join '; ')

    if ($DryRun) {
        $dryC++
        Write-Host "[DRYRUN] $upn => $changesStr" -ForegroundColor Magenta
        $plan.Add([pscustomobject]@{
            Upn=$upn; Params=$updateParams; ChangedFields=$changedList; Changes=$changesStr; Row=$r
        })
        $results.Add([pscustomobject]@{
            Timestamp=$now; CorrelationId=$Global:CorrelationId; UserPrincipalName=$upn
            Status='DryRun'; ChangedFields=$changedList; Changes=$changesStr; Message='Preview only - not written'
        })
    } else {
        try {
            Invoke-GraphWithRetry -Action { Update-MgUser -UserId $upn @updateParams -ErrorAction Stop } | Out-Null

            Write-Host "[OK] $upn => $changesStr" -ForegroundColor Green
            $results.Add([pscustomobject]@{
                Timestamp=$now; CorrelationId=$Global:CorrelationId; UserPrincipalName=$upn
                Status='Success'; ChangedFields=$changedList; Changes=$changesStr; Message='Updated'
            })
        } catch {
            $failC++
            $msg = $_.Exception.Message
            Write-Host "[FAIL] $upn - $msg" -ForegroundColor Red

            try {
                Invoke-GraphWithRetry -Action { Update-MgUser -UserId $upn @updateParams -ErrorAction Stop } | Out-Null
                $okC++; $dryC--
                Write-Host "[OK] $upn => $($p.Changes)" -ForegroundColor Green
                $results.Add([pscustomobject]@{
                    Timestamp=$now; CorrelationId=$Global:CorrelationId; UserPrincipalName=$upn
                    Status='Success'; ChangedFields=$p.ChangedFields; Changes=$p.Changes; Message='Updated (applied from dry-run preview)'
                })
            } catch {
                $failC++; $dryC--
                $msg = $_.Exception.Message
                Write-Host "[FAIL] $upn - $msg" -ForegroundColor Red
                $failed.Add([pscustomobject]@{
                    Timestamp=$now; UserPrincipalName=$upn
                    FirstName=(Get-Cell $r 'First name' $headers); LastName=(Get-Cell $r 'Last name' $headers)
                    DisplayName=(Get-Cell $r 'Display name' $headers); Title=(Get-Cell $r 'Title' $headers)
                    Department=(Get-Cell $r 'Department' $headers); ErrorMessage=$msg; Status='Update Failed'
                })
                $results.Add([pscustomobject]@{
                    Timestamp=$now; CorrelationId=$Global:CorrelationId; UserPrincipalName=$upn
                    Status='Failed'; ChangedFields=$p.ChangedFields; Changes=$p.Changes; Message=$msg
                })
            }
            if ($ai % $heartbeat -eq 0) { Write-Host "[HEARTBEAT] Apply $ai/$($plan.Count) - OK $okC / FAIL $failC" -ForegroundColor DarkGray }
        }
    } else {
        Write-Host "[INFO] Preview kept; nothing written." -ForegroundColor Gray
    }
}

# Final flush of any remaining rows.
if ($results.Count -gt $lastFlush) {
    $slice = $results.GetRange($lastFlush, $results.Count - $lastFlush)
    $slice | Export-Csv -LiteralPath $logFile -NoTypeInformation -Encoding UTF8 -Append
}
if ($failed.Count -gt 0) {
    $failed | Export-Csv -LiteralPath $failedFile -NoTypeInformation -Encoding UTF8
}

$dur = (Get-Date) - $start
Write-Host ""
Write-Host "=== EXECUTION SUMMARY ===" -ForegroundColor Cyan
Write-Host ("Correlation ID  : {0}" -f $Global:CorrelationId)
Write-Host ("Mode            : {0}" -f ($(if ($DryRun) {'DRY-RUN'} else {'LIVE'})))
Write-Host ("Total rows      : {0}" -f $total)
Write-Host ("Updated         : {0}" -f $okC)
Write-Host ("Would-update    : {0}" -f $dryC)
Write-Host ("Skipped         : {0}" -f $skipC)
Write-Host ("Failed          : {0}" -f $failC)
Write-Host ("Duration        : {0:hh\:mm\:ss}" -f $dur)
Write-Host ("Log file        : {0}" -f $logFile)
if ($failed.Count -gt 0) { Write-Host ("Failed CSV      : {0}" -f $failedFile) }
Write-Host "=========================" -ForegroundColor Cyan
