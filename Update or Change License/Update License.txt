<#
============================================================
Bulk Swap Entra ID User Licenses - Phased Execution
============================================================
Description:
- Designed for copy-paste execution into a PowerShell terminal
- Runs in 5 distinct phases with user confirmation
- For each user in the CSV: removes existing license(s) and assigns ONE OR MORE target licenses
- Supports STACKED licenses (e.g. A1 for faculty + A3 for students use benefit)
- Automatically resolves "conflicting services" by keeping the HIGHER tier and disabling
  the lower one (e.g. keeps SharePoint Plan 2 from A3, disables Plan 1 from A1)
- Supports DRY RUN mode to preview changes before making them
- Handles group-assigned licenses gracefully (cannot be removed - logged as Failed)
- Logs all results to a timestamped CSV file with before/after audit trail

Expected CSV columns (case-sensitive headers):
  UserPrincipalName    (or Username, UPN - auto-detected)
Optional:
  UsageLocation        (used to backfill if user doesn't have one - defaults to PH)

Requirements:
- PowerShell 7+
- Microsoft.Graph.Users + Microsoft.Graph.Identity.DirectoryManagement + Microsoft.Graph.Users.Actions
- Admin role: User Administrator or Global Administrator
- Scopes: User.ReadWrite.All, Organization.Read.All

How to use:
- Copy and paste each phase one at a time into your PowerShell terminal
- OR copy and paste the entire script at once
- Recommended: run in DRY RUN mode first (default), review the log, then re-run in LIVE mode
- In Phase 4 Step 2 you may enter several license numbers separated by commas to STACK them.
  The FIRST number you enter is the base SKU; conflicts are resolved automatically by tier.
============================================================
#>

function Get-PlanFamilyInfo {
    param([string]$PlanName)
    if ([string]::IsNullOrWhiteSpace($PlanName)) { return $null }
    $key = $PlanName.Trim().ToUpperInvariant()
    if ($ServicePlanFamilies.ContainsKey($key)) { return $ServicePlanFamilies[$key] }
    return $null
}

# A SKU's tier is derived from the plans it actually carries, never guessed from its
# name - "STANDARDWOFFPACK_IW_FACULTY" (Office 365 A1 for faculty) contains no "A1".
function Get-SkuTierRank {
    param($Sku)
    $max = 0
    foreach ($pl in @($Sku.ServicePlans)) {
        $info = Get-PlanFamilyInfo -PlanName ([string]$pl.ServicePlanName)
        if ($info -and $info.Rank -gt $max) { $max = $info.Rank }
    }
    return $max
}

# Elects one winner per service-plan family across the whole stack (highest tier,
# tie-broken by richer SKU then by stack order) and disables every other member.
# Plans with no known family fall back to same-ServicePlanId de-duplication.
function Resolve-StackConflicts {
    param([object[]]$Stack)

    $disabled  = @{}
    $decisions = [System.Collections.Generic.List[object]]::new()
    if ($null -eq $Stack) { return @{ Disabled = $disabled; Decisions = $decisions } }
    $Stack = @($Stack)
    foreach ($sku in $Stack) { $disabled[$sku.SkuId] = [System.Collections.Generic.List[string]]::new() }
    if ($Stack.Count -le 1) { return @{ Disabled = $disabled; Decisions = $decisions } }

    $skuWeight = @{}
    for ($i = 0; $i -lt $Stack.Count; $i++) {
        $skuWeight[$Stack[$i].SkuId] = [PSCustomObject]@{
            Tier      = (Get-SkuTierRank -Sku $Stack[$i])
            PlanCount = @($Stack[$i].ServicePlans).Count
            Order     = $i
        }
    }

    $entries = [System.Collections.Generic.List[object]]::new()
    foreach ($sku in $Stack) {
        foreach ($pl in @($sku.ServicePlans)) {
            $info = Get-PlanFamilyInfo -PlanName ([string]$pl.ServicePlanName)
            $entries.Add([PSCustomObject]@{
                SkuId    = $sku.SkuId
                SkuPart  = $sku.SkuPartNumber
                PlanId   = [string]$pl.ServicePlanId
                PlanName = [string]$pl.ServicePlanName
                Family   = $(if ($info) { $info.Family } else { "ID:$($pl.ServicePlanId)" })
                Rank     = $(if ($info) { $info.Rank }   else { 0 })
            })
        }
    }

    foreach ($grp in ($entries | Group-Object Family)) {
        $members = @($grp.Group)
        if ($members.Count -le 1) { continue }

        $winner = $members | Sort-Object `
            @{Expression={$_.Rank}; Descending=$true}, `
            @{Expression={$skuWeight[$_.SkuId].Tier}; Descending=$true}, `
            @{Expression={$skuWeight[$_.SkuId].PlanCount}; Descending=$true}, `
            @{Expression={$skuWeight[$_.SkuId].Order}} | Select-Object -First 1

        foreach ($m in $members) {
            if ($m.SkuId -eq $winner.SkuId -and $m.PlanId -eq $winner.PlanId) { continue }
            if (-not $disabled[$m.SkuId].Contains($m.PlanId)) { $disabled[$m.SkuId].Add($m.PlanId) }
            $decisions.Add([PSCustomObject]@{
                Family       = $grp.Name
                KeptPlan     = $winner.PlanName
                KeptOnSku    = $winner.SkuPart
                DisabledPlan = $m.PlanName
                DisabledOn   = $m.SkuPart
                CrossTier    = ($m.PlanId -ne $winner.PlanId)
            })
        }
    }
    return @{ Disabled = $disabled; Decisions = $decisions }
}

    $TargetSkuIds = @($pickedIds)
    $TargetNames  = @($picked | ForEach-Object { $_.SkuPartNumber })

# Helper: sanitize UPN (strips whitespace, invisible chars, normalizes Unicode, lowercases)
function Sanitize-Upn {
    param([string]$Value)
    if ([string]::IsNullOrEmpty($Value)) {
        return @{ Value=''; Changed=$false; Reason=''; Valid=$false; ValidationError='UPN is empty' }
    }
    $original = $Value
    $cleaned = $Value -replace '[\u200B-\u200F\u202A-\u202E\u2066-\u2069\uFEFF]', ''
    $cleaned = $cleaned -replace '[\u0000-\u001F\u007F-\u009F]', ''
    # Unicode normalization: strip diacritics (ñ -> n, é -> e, etc.)
    $cleaned = $cleaned.Normalize([System.Text.NormalizationForm]::FormD)
    $cleaned = $cleaned -replace '\p{Mn}', ''
    $cleaned = $cleaned -replace '[Øø]', 'o'
    $cleaned = $cleaned -replace '[Łł]', 'l'
    $cleaned = $cleaned -replace '[Đđ]', 'd'
    $cleaned = $cleaned -replace '[Þþ]', 'th'
    $cleaned = $cleaned -replace '[ß]', 'ss'
    $cleaned = $cleaned -replace '[Ææ]', 'ae'
    $cleaned = $cleaned -replace '[Œœ]', 'oe'
    $cleaned = $cleaned -replace '[\s\u00A0\u1680\u2000-\u200A\u202F\u205F\u3000]', ''
    $cleaned = $cleaned.ToLowerInvariant()

    $changed = ($cleaned -ne $original)
    $valid = $true
    $verr  = ''
    if ([string]::IsNullOrWhiteSpace($cleaned)) {
        $valid = $false; $verr = 'UPN became empty after sanitization'
    }
    elseif ($cleaned -notmatch '^[^@\s]+@[^@\s]+\.[^@\s]+$') {
        $valid = $false; $verr = "UPN failed format check: '$cleaned'"
    }
    elseif ($cleaned.Length -gt 113) {
        $valid = $false; $verr = "UPN exceeds 113-char limit ($($cleaned.Length) chars)"
    }
    return @{ Value=$cleaned; Changed=$changed; Valid=$valid; ValidationError=$verr }
}

# Helper: retry Graph calls on 429/5xx with backoff
function Invoke-GraphWithRetry {
    param([Parameter(Mandatory)][scriptblock]$Action, [int]$MaxAttempts = 5, [int]$BaseDelaySeconds = 5)
    for ($attempt = 1; $attempt -le $MaxAttempts; $attempt++) {
        try { return & $Action }
        catch {
            $isLast = ($attempt -eq $MaxAttempts)
            $msg = $_.Exception.Message
            $status = $null
            if ($_.Exception.Response -and $_.Exception.Response.StatusCode) {
                $status = [int]$_.Exception.Response.StatusCode
            }
            $isRetryable = ($status -eq 429) -or ($status -ge 500 -and $status -lt 600) -or ($msg -match 'throttl|TooManyRequests|timed out|timeout')
            if (-not $isRetryable -or $isLast) { throw }
            $delay = $BaseDelaySeconds * [math]::Pow(2, $attempt - 1)
            if ($_.Exception.Response -and $_.Exception.Response.Headers) {
                $retryAfter = $_.Exception.Response.Headers['Retry-After']
                if ($retryAfter -and ($retryAfter -as [int])) { $delay = [int]$retryAfter }
            }
            Write-Host "    Throttled/transient (status=$status). Retrying in $delay s... ($attempt/$MaxAttempts)" -ForegroundColor DarkYellow
            Start-Sleep -Seconds $delay
        }
    }
}

# Helper: convert a list of SKU IDs to a semicolon-joined string of part numbers (for logging)
function Format-SkuList {
    param([string[]]$SkuIds)
    if (-not $SkuIds -or $SkuIds.Count -eq 0) { return "(none)" }
    $names = foreach ($id in $SkuIds) {
        if ($SkuIdToPartNumber.ContainsKey($id)) { $SkuIdToPartNumber[$id] } else { $id }
    }
    return ($names -join '; ')
}

            LicensesAdded         = $AddedStr
            LicensesRemoved       = $RemovedStr
            GroupAssignedLicenses = $BlockedStr
            DisabledPlans         = $DisabledPlanStr
            Status                = "WouldChange"
            Message               = $dryMsg
        })
        $Counters.WouldChange++
        continue
    }

    # LIVE MODE: check UsageLocation and backfill if needed
    $NeedsUsageLocation = ($NeedsTargetAdded -and [string]::IsNullOrWhiteSpace($ExistingUser.UsageLocation))
    if ($NeedsUsageLocation) {
        $ULocToSet = if (-not [string]::IsNullOrWhiteSpace($CsvUsageLocation)) { $CsvUsageLocation } else { $DefaultUsageLocation }
        try {
            Invoke-GraphWithRetry { Update-MgUser -UserId $ExistingUser.Id -UsageLocation $ULocToSet -ErrorAction Stop }
            Write-Host "[Row $RowNumber]   UsageLocation backfilled: '$ULocToSet'" -ForegroundColor Yellow
        }
        catch {
            $errMsg = $_.Exception.Message
            Write-Host "[Row $RowNumber]   FAILED to set UsageLocation: $errMsg" -ForegroundColor Red
            $Log.Add([PSCustomObject]@{
                Timestamp             = (Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
                Row                   = $RowNumber
                UserPrincipalName     = $UPN
                RunMode               = $RunMode
                Strategy              = $RemovalStrategy
                OldLicenses           = $OldLicensesStr
                NewLicenses           = $OldLicensesStr
                LicensesAdded         = ""
                LicensesRemoved       = ""
                GroupAssignedLicenses = $BlockedStr
                DisabledPlans         = ""
                Status                = "Failed"
                Message               = "UsageLocation backfill failed: $errMsg"
            })
            $Counters.Failed++
            continue
        }
    }

    # Execute the swap (atomic add+remove in one Set-MgUserLicense call).
    # $AddParams already carries DisabledPlans per SKU, so a stacked assignment lands
    # with its conflicts resolved in the same request.
    try {
        $RemoveParams = @($LicensesToRemove)

        Write-Host "[Row $RowNumber]   Swapping: remove [$RemovedStr] + add [$AddedStr]" -ForegroundColor Cyan
        if ($DisabledPlanStr) {
            Write-Host "[Row $RowNumber]   Disabling conflicting services -> $DisabledPlanStr" -ForegroundColor DarkYellow
        }
        try {
            Invoke-GraphWithRetry {
                Set-MgUserLicense -UserId $ExistingUser.Id `
                    -AddLicenses $AddParams `
                    -RemoveLicenses $RemoveParams `
                    -ErrorAction Stop | Out-Null
            }
        }
        catch {
            # Graph can refuse a combined add+remove when the incumbent license conflicts
            # with the incoming stack. Retry as remove-first, then add.
            $cMsg = $_.Exception.Message
            if ($_.ErrorDetails -and $_.ErrorDetails.Message) { $cMsg = "$cMsg $($_.ErrorDetails.Message)" }
            if ($cMsg -match '(?i)conflict|mutually exclusive') {
                Write-Host "[Row $RowNumber]   Conflict on combined call - retrying as remove-then-add..." -ForegroundColor DarkYellow
                Invoke-GraphWithRetry {
                    if ($RemoveParams.Count -gt 0) {
                        Set-MgUserLicense -UserId $ExistingUser.Id -AddLicenses @() -RemoveLicenses $RemoveParams -ErrorAction Stop | Out-Null
                    }
                    Set-MgUserLicense -UserId $ExistingUser.Id -AddLicenses $AddParams -RemoveLicenses @() -ErrorAction Stop | Out-Null
                }
            } else { throw }
        }
        Write-Host "[Row $RowNumber]   SUCCESS" -ForegroundColor Green

        $successMsg = "Removed [$RemovedStr]; added [$AddedStr]"
        if ($BlockedByGroup.Count -gt 0) {
            $successMsg += " | Note: $BlockedStr blocked by group assignment"
        }
        $Log.Add([PSCustomObject]@{
            Timestamp             = (Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
            Row                   = $RowNumber
            UserPrincipalName     = $UPN
            RunMode               = $RunMode
            Strategy              = $RemovalStrategy
            OldLicenses           = $OldLicensesStr
            NewLicenses           = $NewLicensesStr
            LicensesAdded         = $AddedStr
            LicensesRemoved       = $RemovedStr
            GroupAssignedLicenses = $BlockedStr
            DisabledPlans         = $DisabledPlanStr
            Status                = "Success"
            Message               = $successMsg
        })
        $Counters.Success++
    }
    catch {
        $errMsg = $_.Exception.Message
        if ($_.ErrorDetails -and $_.ErrorDetails.Message) {
            $errMsg = "$errMsg | Details: $($_.ErrorDetails.Message)"
        }
        # Add specific hint for common error patterns
        if ($errMsg -match 'mutually exclusive') {
            $errMsg += " | HINT: Try strategy 'B' (remove ALL) or 'C' (remove EDU) to resolve service plan conflicts."
        }
        if ($errMsg -match 'group|assigned by group') {
            $errMsg += " | HINT: Some licenses are assigned via group membership and cannot be removed via user API."
        }
        Write-Host "[Row $RowNumber]   FAILED swap: $errMsg" -ForegroundColor Red
        $Log.Add([PSCustomObject]@{
            Timestamp             = (Get-Date -Format 'yyyy-MM-dd HH:mm:ss')
            Row                   = $RowNumber
            UserPrincipalName     = $UPN
            RunMode               = $RunMode
            Strategy              = $RemovalStrategy
            OldLicenses           = $OldLicensesStr
            NewLicenses           = $OldLicensesStr
            LicensesAdded         = ""
            LicensesRemoved       = ""
            GroupAssignedLicenses = $BlockedStr
            DisabledPlans         = $DisabledPlanStr
            Status                = "Failed"
            Message               = "Swap failed: $errMsg"
        })
        $Counters.Failed++
    }
}

# Export log to CSV
try {
    $Log | Export-Csv -Path $LogPath -NoTypeInformation -Encoding UTF8 -ErrorAction Stop
    Write-Host ""
    Write-Host "Log file saved to:" -ForegroundColor Green
    Write-Host "  $LogPath" -ForegroundColor Green
}
catch {
    Write-Host "ERROR: Failed to export log CSV: $($_.Exception.Message)" -ForegroundColor Red
}

Write-Host " Summary ($RunMode)" -ForegroundColor Cyan
Write-Host "------------------------------------------------------------" -ForegroundColor Cyan

if ($IsDryRun) {
    Write-Host (" Would change        : {0}" -f $Counters.WouldChange) -ForegroundColor Yellow
} else {

}

Write-Host (" Skipped             : {0}" -f $Counters.Skipped)    -ForegroundColor Yellow
Write-Host (" End Time            : {0}" -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'))
Write-Host "============================================================" -ForegroundColor Cyan

if ($IsDryRun -and $Counters.WouldChange -gt 0) {
    Write-Host ""
    Write-Host "This was a DRY RUN. To apply these changes, re-run the script and choose Run mode [2] LIVE." -ForegroundColor Yellow
}
